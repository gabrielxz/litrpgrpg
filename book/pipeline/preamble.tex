% A spot illustration anchored to the page: it runs from its column out to the
% outer trim edge, with an ink rule on the inner edge and one cyan tick. Called
% inside a minipage whose width is the column; the overflow is the outer margin.
\newlength{\gbspotreach}
\setlength{\gbspotreach}{0.38in}   % how far past the column the art reaches into the outer margin (0.7in), leaving air before the trim
\newcommand{\gbspotart}[1]{%
  \begin{tikzpicture}[baseline=(img.north)]
    \node[anchor=north west, inner sep=0pt] (img) at (0,0)
      {\includegraphics[width=\dimexpr\linewidth+\gbspotreach\relax]{#1}};
    \draw[gbink, line width=0.9pt] ([xshift=-7pt]img.north west) -- ([xshift=-7pt]img.south west);
    \draw[gbcyan, line width=1.6pt] ([xshift=-7pt, yshift=-10pt]img.north west) -- ([xshift=-7pt, yshift=-30pt]img.north west);
  \end{tikzpicture}%
}

\newcommand{\fullpageart}[1]{%
  \clearpage%
  \thispagestyle{empty}%
  \sbox\gbartbox{\includegraphics[width=\paperwidth]{#1}}%
  \ifdim\ht\gbartbox<\paperheight
    \sbox\gbartbox{\includegraphics[height=\paperheight]{#1}}%
  \fi
  \begin{tikzpicture}[remember picture, overlay]
    \clip (current page.south west) rectangle (current page.north east);
    \node[anchor=center, inner sep=0pt] at (current page.center) {\usebox\gbartbox};
  \end{tikzpicture}%
  \clearpage%
}

% Epigraph: an in-world quotation set under a chapter title. No box; a
% right-set block, italic, with the source in upright small type beneath.
% Markdown: ::: epigraph / quote paragraph / source paragraph / :::
% The Lua filter maps the class to this environment and wraps the last
% paragraph in gbepigraphsource.
\newenvironment{bookepigraph}{%
  \par\vspace{4pt}%
  \begin{flushright}%
  \begin{minipage}{0.78\linewidth}%
  \raggedright\itshape\small\setlength{\parskip}{3pt}%
}{%
  \end{minipage}%
  \end{flushright}%
  \par\vspace{4pt}%
}
\newenvironment{gbepigraphsource}{\par\nopagebreak\raggedleft\upshape\footnotesize\color{gbslate}}{\par}
