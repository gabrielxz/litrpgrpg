\usepackage{fontspec}

% Body, sans, and mono are declared in metadata.yaml (mainfont, sansfont,
% monofont): EB Garamond, Alegreya Sans, JetBrains Mono. The template's own
% font block would override a \setmainfont placed here.

% Heading font: Alegreya Black — display serif with weight
\newfontfamily\headingfont{Alegreya}[
  Extension=.otf,
  UprightFont=*-Black,
  ItalicFont=*-BlackItalic
]

% Apply heading font to all KOMA-Script "disposition" elements
% (chapters, sections, subsections, paragraph titles)
\addtokomafont{disposition}{\headingfont}
\addtokomafont{chapter}{\headingfont}
\addtokomafont{section}{\headingfont}
\addtokomafont{subsection}{\headingfont}

% Chapter titles carry an interrupted rule: a long dark segment, a gap, one short cyan segment.
\makeatletter
\renewcommand*{\chapterlinesformat}[3]{%
  \@hangfrom{#2}{#3}\par\nobreak
  \vskip 6pt
  \noindent\begin{tikzpicture}
    \draw[gbink, line width=0.9pt] (0,0) -- (0.60\textwidth,0);
    \draw[gbcyan, line width=1.4pt] (0.645\textwidth,0) -- (0.70\textwidth,0);
  \end{tikzpicture}\par
}
\makeatother

% Table Kit appendix: the kit PDF (built by `make kit`) is inserted whole,
% so the sheet is authored once in book/kit/table-kit.html and cannot drift.
\usepackage{pdfpages}

\newsavebox\gbartbox
\newcommand{\fullpageart}[1]{%
  \clearpage%
  \thispagestyle{empty}%
  \sbox\gbartbox{\includegraphics[width=\paperwidth]{#1}}%
  \ifdim\ht\gbartbox<\paperheight
    \sbox\gbartbox{\includegraphics[height=\paperheight]{#1}}%
  \fi
  \begin{tikzpicture}[remember picture, overlay]
    \clip (current page.south west) rectangle (current page.north east);
    \node[anchor=center, inner sep=0pt] at (current page.center) {\usebox\gbartbox};
  \end{tikzpicture}%
  \clearpage%
}

% -----------------------------------------------------
% Callout box environments (tcolorbox-based)

\usepackage[most]{tcolorbox}

% System Voice — dark box with cyan rule, italic body
% For in-fiction System messages: "Consciousness anchored. Status: Integrated."
\newtcolorbox{systemvoice}{
  enhanced,
  frame hidden,
  colback=gbink,
  coltext=gbbone,
  coltitle=gbcyanlight,
  colbacktitle=gbink,
  fonttitle=\footnotesize\sffamily\bfseries,
  title={THE SYSTEM},
  fontupper=\sffamily,
  boxrule=0pt,
  arc=0pt,
  outer arc=0pt,
  left=14pt,
  right=14pt,
  top=6pt,
  bottom=12pt,
  before skip=10pt,
  after skip=10pt,
  overlay={
    \draw[gbcyanlight, line width=1.1pt] ([yshift=-40pt]frame.north west) -- (frame.north west) -- ([xshift=60pt]frame.north west);
    \draw[gbcyanlight, line width=1.1pt] ([xshift=-60pt]frame.south east) -- (frame.south east) -- ([yshift=40pt]frame.south east);
  }
}

% Read Aloud — boxed narration the GM says out loud, titled so it can be
% found while scanning a page mid-session. White ground and a heavy dark
% frame so it survives grayscale e-ink. Deliberately NOT breakable: a
% read-aloud passage that splits across a page turn cannot be read cold,
% so keep each block short enough to sit on one page.
\newtcolorbox{readaloud}{
  enhanced,
  frame hidden,
  colback=gbbonedeep,
  coltitle=gbink,
  colbacktitle=gbbonedeep,
  boxrule=0pt,
  arc=0pt,
  outer arc=0pt,
  fonttitle=\footnotesize\sffamily\bfseries,
  title={READ ALOUD},
  parbox=false,
  before upper={\setlength{\parindent}{0pt}\setlength{\parskip}{7pt}},
  left=14pt,
  right=14pt,
  top=6pt,
  bottom=12pt,
  before skip=12pt,
  after skip=12pt,
  overlay={
    \draw[gbink, line width=0.9pt] (frame.south west) -- (frame.north west) -- ([xshift=-70pt]frame.north east);
    \draw[gbink, line width=0.9pt] ([xshift=-30pt]frame.north east) -- (frame.north east) -- ([yshift=-30pt]frame.north east);
    \draw[gbink, line width=0.9pt] (frame.south west) -- ([xshift=90pt]frame.south west);
    \draw[gbcyan, line width=1.4pt] ([xshift=104pt]frame.south west) -- ([xshift=128pt]frame.south west);
  }
}

% Running header fix: with scrbook + oneside, chapters only \markright,
% so \leftmark keeps the table of contents' "Contents" mark for the whole
% book. Mark both sides from the chapter so header-left (\leftmark in
% metadata.yaml) tracks the current chapter name.
\AtBeginDocument{%
  \renewcommand{\chaptermark}[1]{\markboth{#1}{}}%
}
